document.addEventListener("DOMContentLoaded", () => { const scene = document.querySelector("a-scene"); const videos = []; for (let i = 0; i < 30; i++) { videos.push(document.querySelector("#video" + i)); } let activeTarget = null; let activeIndex = -1; // ========================= // AR READY // ========================= scene.addEventListener("arReady", () => { console.log("AR READY"); }); // ========================= // TARGET FOUND // ========================= scene.addEventListener("targetFound", async (e) => { const target = e.target; const data = target.getAttribute("mindar-image-target"); const index = data.targetIndex; console.log("TARGET FOUND:", index); activeTarget = target; activeIndex = index; // ------------------------- // فقط ویدیوهای دیگر را متوقف و مخفی کن // ------------------------- const allTargets = scene.querySelectorAll( "[mindar-image-target]" ); allTargets.forEach((t) => { const plane = t.querySelector("a-video"); if (plane && t !== target) { plane.setAttribute("visible", false); // اطمینان از مخفی شدن واقعی if (plane.object3D) { plane.object3D.visible = false; } } }); // ------------------------- // همه ویدیوها را متوقف کن // ------------------------- videos.forEach((video, i) => { if (!video) return; if (i !== index) { video.pause(); try { video.currentTime = 0; } catch (err) {} } }); // ------------------------- // ویدیوی مربوط به تارگت // ------------------------- const video = videos[index]; if (!video) { console.log("VIDEO NOT FOUND:", index); return; } console.log( "VIDEO:", index, "READY STATE:", video.readyState ); // از اول ویدیو try { video.currentTime = 0; } catch (err) {} // صدا مثل نسخه قبلی video.muted = false; video.volume = 1; // ------------------------- // اگر ویدیو هنوز داده ندارد // ------------------------- if (video.readyState < 2) { console.log("WAITING FOR VIDEO DATA..."); await new Promise((resolve) => { let finished = false; const done = () => { if (finished) return; finished = true; video.removeEventListener( "loadeddata", done ); video.removeEventListener( "canplay", done ); console.log( "VIDEO DATA READY:", video.readyState ); resolve(); }; video.addEventListener( "loadeddata", done, { once: true } ); video.addEventListener( "canplay", done, { once: true } ); // اگر قبلاً آماده شد if (video.readyState >= 2) { done(); } // حداکثر 10 ثانیه setTimeout(done, 10000); }); } // ------------------------- // بررسی اینکه هنوز همین تارگت است // ------------------------- if ( activeTarget !== target || activeIndex !== index ) { console.log("TARGET CHANGED - CANCEL"); return; } // ------------------------- // حالا پلین همین تارگت را نشان بده // ------------------------- const plane = target.querySelector("a-video"); if (!plane) { console.log("A-VIDEO NOT FOUND:", index); return; } plane.setAttribute("visible", true); if (plane.object3D) { plane.object3D.visible = true; } console.log("VIDEO PLANE VISIBLE:", index); // ------------------------- // PLAY // ------------------------- try { await video.play(); console.log( "VIDEO PLAYING:", index ); } catch (error) { console.log( "PLAY WITH SOUND FAILED:", error ); // تلاش دوم بدون صدا video.muted = true; try { await video.play(); console.log( "VIDEO PLAYING MUTED:", index ); } catch (error2) { console.log( "VIDEO PLAY FAILED:", error2 ); } } }); // ========================= // TARGET LOST // ========================= scene.addEventListener("targetLost", (e) => { const target = e.target; const data = target.getAttribute("mindar-image-target"); const index = data.targetIndex; console.log("TARGET LOST:", index); const video = videos[index]; if (video) { video.pause(); try { video.currentTime = 0; } catch (err) {} } // فوراً ویدیو را مخفی کن const plane = target.querySelector("a-video"); if (plane) { plane.setAttribute("visible", false); if (plane.object3D) { plane.object3D.visible = false; } } // فقط اگر همین تارگت فعال بود if (activeTarget === target) { activeTarget = null; activeIndex = -1; } }); });
